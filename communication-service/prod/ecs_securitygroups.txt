# ECS SG's

# Event Service SG : Connect to Public LB
resource "aws_security_group" "ecs_sg_event" {
  name   = "${var.ProjectName}-ecs-sg-event-${var.env}"
  vpc_id = var.vpcId

  ingress {
    from_port       = var.event_ecs_container_port
    to_port         = var.event_ecs_container_port
    protocol        = "tcp"
    security_groups = [aws_security_group.sec-grps["Alb-SG"].id]
  }

  egress {
    from_port   = 0
    to_port     = 0
    protocol    = "-1"
    cidr_blocks = ["0.0.0.0/0"]
  }

  tags = {
    Name        = "${var.ProjectName}-ecs-sg-event-${var.env}"
    ProjectName = var.ProjectName
    CreatedBy   = var.CreatedBy
  }
}
